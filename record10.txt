2.create a procedure to display invoice details of a particular customer.

create procedure invoice_detail(in c_id int) begin select customer.cust_id,customer.cust_name,product.prd_id,invoice_master.inv_date,invoice_item.iv_id,invoice_item.quantity from customer join invoice_master on customer.cust_id=invoice_master.cust_id join  invoice_item on invoice_master.iv_id=invoice_item.iv_id join product on invoice_item.prd_id=product.prd_id where customer.cust_id=c_id; end
--------------
