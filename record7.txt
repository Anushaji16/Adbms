2. display the customer name,product name invoice date of purchased item 'pen'

mysql> SELECT customer.cust_name, product.prd_name, invoice_master.inv_date  FROM customer  JOIN invoice_master ON customer.cust_id = invoice_master.cust_id  JO
IN invoice_item  ON invoice_master.iv_id = invoice_item.iv_id  JOIN product
 ON invoice_item.prd_id = product.prd_id  WHERE product.prd_name = 'pencil';
